mainFrame.BorderSizePixel = 0
mainFrame.Active = true

mainFrame.Parent = screenGui

local uiCorner = Instance.new("UICorner")
uiCorner.CornerRadius = UDim.new(0, 10)
uiCorner.Parent = mainFrame

-- Campo de texto para poner el nombre o ID

btnCorner.CornerRadius = UDim.new(0, 6)
btnCorner.Parent = copyBtn

-- Botón de Minimizar / Maximizar

minimizeBtn.Size = UDim2.new(0.42, 0, 0, 30)
minimizeBtn.Position = UDim2.new(0.05, 0, 0, 130)

-- Botón para Cerrar / Destruir la GUI por completo
local closeBtn = Instance.new("TextButton")
closeBtn.Size = UDim2.new(0.42, 0, 0, 30)
closeBtn.Position = UDim2.new(0.53, 0, 0, 130)
closeBtn.BackgroundColor3 = Color3.fromRGB(150, 40, 40)
closeBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
closeBtn.TextSize = 11
closeBtn.Font = Enum.Font.SourceSansBold
closeBtn.Text = "Cerrar"

local closeCorner = Instance.new("UICorner")
closeCorner.CornerRadius = UDim.new(0, 6)
closeCorner.Parent = closeBtn

-- Lógica para copiar la descripción del avatar por ID
local function applyAvatar(targetUserId)
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        pcall(function()
            local content = Players:GetHumanoidDescriptionFromUserId(targetUserId)
            humanoid:ApplyDescription(content)
        end)
    end
end

-- Evento del botón Copiar
copyBtn.MouseButton1Click:Connect(function()
    local input = textBox.Text
    local targetId = tonumber(input)
    
    if targetId then
        applyAvatar(targetId)
    else
        -- Si escribieron un nombre en lugar de un número, busca el UserId automáticamente
        pcall(function()
            local id = Players:GetUserIdFromNameAsync(input)
            if id then
                applyAvatar(id)
            end
        end)
    end
end)

-- Lógica de Minimizar
local minimized = false
minimizeBtn.MouseButton1Click:Connect(function()
    minimized = not minimized
    if minimized then
        mainFrame.Size = UDim2.new(0, 240, 0, 35)
        minimizeBtn.Text = "Restaurar"
        textBox.Visible = false
        copyBtn.Visible = false
        closeBtn.Visible = false
    else
        mainFrame.Size = UDim2.new(0, 240, 0, 180)
        minimizeBtn.Text = "Minimizar"
        textBox.Visible = true
        copyBtn.Visible = true
        closeBtn.Visible = true
    end
end)

-- Lógica de Cerrar por completo
closeBtn.MouseButton1Click:Connect(function()
    screenGui:Destroy()
end)

print("[+] Script de clonar avatar cargado con éxito.")
