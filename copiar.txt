-- Función compatible con Delta para aplicar la descripción sin bloqueos
local function applyAvatarToMe(targetId)
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    -- Usamos pcall protegido compatible con el entorno iOS de Delta
    pcall(function()
        local desc = Players:GetHumanoidDescriptionFromUserId(targetId)
        if desc then
            humanoid:ApplyDescriptionReset(desc)
        end
    end)
end

copyBtn.MouseButton1Click:Connect(function()
    local text = textBox.Text
    local id = tonumber(text)
    
    if id then
        applyAvatarToMe(id)
    else
        pcall(function()
            local resolvedId = Players:GetUserIdFromNameAsync(text)
            if resolvedId then
                applyAvatarToMe(resolvedId)
            end
        end)
    end
end)

local minimized = false
minimizeBtn.MouseButton1Click:Connect(function()
    minimized = not minimized
    if minimized then
        mainFrame.Size = UDim2.new(0, 220, 0, 30)
        minimizeBtn.Text = "Restaurar"
        textBox.Visible = false
        copyBtn.Visible = false
        closeBtn.Visible = false
    else
        mainFrame.Size = UDim2.new(0, 220, 0, 160)
        minimizeBtn.Text = "Minimizar"
        textBox.Visible = true
        copyBtn.Visible = true
        closeBtn.Visible = true
    end
end)

closeBtn.MouseButton1Click:Connect(function()
    screenGui:Destroy()
end)
