-- Función alternativa segura usando HumanoidDescription directo por ID
local function copyTargetAvatar(userId)
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    task.spawn(function()
        local success, desc = pcall(function()
            return Players:GetHumanoidDescriptionFromUserId(userId)
        end)
        
        if success and desc then
            -- Aplica la descripción completa forzando las propiedades del modelo
            local currentDesc = humanoid:GetAppliedDescription()
            if currentDesc then
                currentDesc.Shirt = desc.Shirt
                currentDesc.Pants = desc.Pants
                currentDesc.Graphic = desc.Graphic
                currentDesc.Face = desc.Face
                humanoid:ApplyDescription(currentDesc)
            else
                humanoid:ApplyDescription(desc)
            end
        else
            warn("No se pudo obtener la descripción del usuario.")
        end
    end)
end

copyBtn.MouseButton1Click:Connect(function()
    local input = textBox.Text
    local targetId = tonumber(input)
    
    if targetId then
        copyTargetAvatar(targetId)
    else
        local success, id = pcall(function()
            return Players:GetUserIdFromNameAsync(input)
        end)
        if success and id then
            copyTargetAvatar(id)
        end
    end
end)

    minimized = not minimized
    if minimized then
        mainFrame.Size = UDim2.new(0, 240, 0, 35)
        minimizeBtn.Text = "Restaurar"
        textBox.Visible = false

        closeBtn.Visible = false
    else
        mainFrame.Size = UDim2.new(0, 240, 0, 180)
        minimizeBtn.Text = "Minimizar"
        textBox.Visible = true
