-- Función para aplicar mediante consulta directa compatible con ejecutores móviles
local function applyClothing(userId)
    local character = player.Character
    if not character then return end
    
    task.spawn(function()
        -- Petición segura a la API pública de avatares de Roblox
        local success, response = pcall(function()
            return game:HttpGet("https://avatar.roblox.com/v1/users/"..tostring(userId).."/avatar")
        end)
        
        if success and response then
            local data = HttpService:JSONDecode(response)
            if data and data.assets then
                local shirtId, pantsId, tShirtId = nil, nil, nil
                
                for _, asset in ipairs(data.assets) do
                    if asset.assetType then
                        if asset.assetType.name == "Shirt" then
                            shirtId = asset.id
                        elseif asset.assetType.name == "Pants" then
                            pantsId = asset.id
                        elseif asset.assetType.name == "TShirt" then
                            tShirtId = asset.id
                        end
                    end
                end
                
                local humanoid = character:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    -- Buscar o crear objetos de ropa en el personaje
                    local shirt = character:FindFirstChildOfClass("Shirt") or Instance.new("Shirt", character)
                    local pants = character:FindFirstChildOfClass("Pants") or Instance.new("Pants", character)
                    
                    if shirtId then
                        shirt.ShirtTemplate = "rbxassetid://"..tostring(shirtId)
                    end
                    if pantsId then
                        pants.PantsTemplate = "rbxassetid://"..tostring(pantsId)
                    end
                end
            end
        end
    end)
end

copyBtn.MouseButton1Click:Connect(function()
    local text = textBox.Text
    local id = tonumber(text)
    
    if id then
        applyClothing(id)
    else
        pcall(function()
            local resolvedId = Players:GetUserIdFromNameAsync(text)
            if resolvedId then
                applyClothing(resolvedId)
            end
        end)
    end
end)
