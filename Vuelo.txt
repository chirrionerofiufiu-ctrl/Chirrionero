--[==[ 
    Panel GUI de Vuelo Optimizado para Celular - JJS
]==]--

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local player = Players.LocalPlayer

local flying = false
local speed = 45
local bodyVelocity, bodyGyro
local currentAnimTrack

-- IDs de animaciones adaptadas para móvil
local anims = {
    {name = "Estilo Pro / Aura", moveId = "rbxassetid://507768375", idleId = "rbxassetid://507768434"},
    {name = "Postura Amenazante", moveId = "rbxassetid://507768388", idleId = "rbxassetid://507768375"}
}
local currentSelection = anims[1]

-- Interfaz Gráfica (GUI) Compacta para Celular
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "MobileFlightGui"
screenGui.ResetOnSpawn = false
screenGui.Parent = player:WaitForChild("PlayerGui")

local mainFrame = Instance.new("Frame")
mainFrame.Size = UDim2.new(0, 220, 0, 240) -- Tamaño más pequeño y cómodo
mainFrame.Position = UDim2.new(0.02, 0, 0.25, 0)
mainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
mainFrame.BorderSizePixel = 0
mainFrame.Active = true
mainFrame.Draggable = true
mainFrame.Parent = screenGui

local uiCorner = Instance.new("UICorner")
uiCorner.CornerRadius = UDim.new(0, 10)
uiCorner.Parent = mainFrame

local titleLabel = Instance.new("TextLabel")
titleLabel.Size = UDim2.new(1, 0, 0, 30)
titleLabel.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
titleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
titleLabel.TextSize = 13
titleLabel.Font = Enum.Font.SourceSansBold
titleLabel.Text = "Vuelo Pro - Mobile"
titleLabel.Parent = mainFrame

local titleCorner = Instance.new("UICorner")
titleCorner.CornerRadius = UDim.new(0, 10)
titleCorner.Parent = titleLabel

local function playAnimation(id)
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        local animator = humanoid:FindFirstChildOfClass("Animator") or Instance.new("Animator", humanoid)
        if currentAnimTrack then currentAnimTrack:Stop() end
        local anim = Instance.new("Animation")
        anim.AnimationId = id
        currentAnimTrack = animator:LoadAnimation(anim)
        currentAnimTrack:Play()
    end
end

local function startFlight()
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    local rootPart = character.HumanoidRootPart
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    
    flying = true
    
    -- Evita el bug de congelamiento en móvil aplicando fuerzas estables
    playAnimation(currentSelection.moveId)

    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.Velocity = Vector3.new(0, 0, 0)
    bodyVelocity.MaxForce = Vector3.new(400000, 400000, 400000)
    bodyVelocity.Parent = rootPart

    bodyGyro = Instance.new("BodyGyro")
    bodyGyro.MaxTorque = Vector3.new(400000, 400000, 400000)
    bodyGyro.CFrame = rootPart.CFrame
    bodyGyro.Parent = rootPart

    task.spawn(function()
        while flying and character and character:Parent() do
            local camera = workspace.CurrentCamera
            local moveDir = Vector3.new(0, 0, 0)
            
            -- Movimiento adaptado para controles táctiles / cámara
            if UserInputService:IsKeyDown(Enum.KeyCode.W) or UserInputService:IsKeyDown(Enum.KeyCode.Up) then 
                moveDir = moveDir + camera.CFrame.LookVector 
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) or UserInputService:IsKeyDown(Enum.KeyCode.Down) then 
                moveDir = moveDir - camera.CFrame.LookVector 
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) or UserInputService:IsKeyDown(Enum.KeyCode.Left) then 
                moveDir = moveDir - camera.CFrame.RightVector 
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) or UserInputService:IsKeyDown(Enum.KeyCode.Right) then 
                moveDir = moveDir + camera.CFrame.RightVector 
            end

            bodyVelocity.Velocity = moveDir * speed
            bodyGyro.CFrame = camera.CFrame
            RunService.RenderStepped:Wait()
        end
    end)
end

local function stopFlight()
    flying = false
    if bodyVelocity then bodyVelocity:Destroy() end
    if bodyGyro then bodyGyro:Destroy() end
    if currentAnimTrack then currentAnimTrack:Stop() end
end

-- Botón táctil para Activar/Desactivar Vuelo
local toggleBtn = Instance.new("TextButton")
toggleBtn.Size = UDim2.new(0.9, 0, 0, 30)
toggleBtn.Position = UDim2.new(0.05, 0, 0, 40)
toggleBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
toggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
toggleBtn.TextSize = 13
toggleBtn.Font = Enum.Font.SourceSansBold
toggleBtn.Text = "Activar / Apagar Vuelo"
toggleBtn.Parent = mainFrame

toggleBtn.MouseButton1Click:Connect(function()
    if flying then
        stopFlight()
        toggleBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    else
        startFlight()
        toggleBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 100)
    end
end)

-- Selector de animación táctil
local animBtn = Instance.new("TextButton")
animBtn.Size = UDim2.new(0.9, 0, 0, 28)
animBtn.Position = UDim2.new(0.05, 0, 0, 80)
animBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
animBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
animBtn.TextSize = 12
animBtn.Font = Enum.Font.SourceSans
animBtn.Text = "Cambiar Estilo Pro"
animBtn.Parent = mainFrame

animBtn.MouseButton1Click:Connect(function()
    if currentSelection == anims[1] then
        currentSelection = anims[2]
        animBtn.Text = "Estilo: Amenazante"
    else
        currentSelection = anims[1]
        animBtn.Text = "Estilo: Pro / Aura"
    end
    if flying then
        stopFlight()
        startFlight()
    end
end)

-- Controles de Velocidad Táctiles
local speedLabel = Instance.new("TextLabel")
speedLabel.Size = UDim2.new(0.9, 0, 0, 20)
speedLabel.Position = UDim2.new(0.05, 0, 0, 120)
speedLabel.BackgroundTransparency = 1
speedLabel.TextColor3 = Color3.fromRGB(200, 200, 200)
speedLabel.TextSize = 12
speedLabel.Font = Enum.Font.SourceSansBold
speedLabel.Text = "Velocidad: " .. speed
speedLabel.Parent = mainFrame

local minusBtn = Instance.new("TextButton")
minusBtn.Size = UDim2.new(0.42, 0, 0, 28)
minusBtn.Position = UDim2.new(0.05, 0, 0, 145)
minusBtn.BackgroundColor3 = Color3.fromRGB(140, 40, 40)
minusBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
minusBtn.Text = "Vel -"
minusBtn.Parent = mainFrame

minusBtn.MouseButton1Click:Connect(function()
    speed = math.max(15, speed - 10)
    speedLabel.Text = "Velocidad: " .. speed
end)

local plusBtn = Instance.new("TextButton")
plusBtn.Size = UDim2.new(0.42, 0, 0, 28)
plusBtn.Position = UDim2.new(0.53, 0, 0, 145)
plusBtn.BackgroundColor3 = Color3.fromRGB(40, 120, 40)
plusBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
plusBtn.Text = "Vel +"
plusBtn.Parent = mainFrame

plusBtn.MouseButton1Click:Connect(function()
    speed = speed + 10
    speedLabel.Text = "Velocidad: " .. speed
end)

-- Botón para cerrar / ocultar GUI si estorba
local closeBtn = Instance.new("TextButton")
closeBtn.Size = UDim2.new(0.9, 0, 0, 25)
closeBtn.Position = UDim2.new(0.05, 0, 0, 195)
closeBtn.BackgroundColor3 = Color3.fromRGB(80, 80, 80)
closeBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
closeBtn.TextSize = 11
closeBtn.Text = "Minimizar / Ocultar"
closeBtn.Parent = mainFrame

local minimized = false
closeBtn.MouseButton1Click:Connect(function()
    minimized = not minimized
    if minimized then
        mainFrame.Size = UDim2.new(0, 220, 0, 35)
        closeBtn.Text = "Maximizar Panel"
        -- Oculta los demás elementos visuales al minimizar
        toggleBtn.Visible = false
        animBtn.Visible = false
        speedLabel.Visible = false
        minusBtn.Visible = false
        plusBtn.Visible = false
    else
        mainFrame.Size = UDim2.new(0, 220, 0, 240)
        closeBtn.Text = "Minimizar / Ocultar"
        toggleBtn.Visible = true
        animBtn.Visible = true
        speedLabel.Visible = true
        minusBtn.Visible = true
        plusBtn.Visible = true
    end
end)

print("[+] Panel optimizado para móvil cargado.")
