--[==[ 
    Panel GUI de Vuelo Estilo Pro + Idle Animado para JJS
]==]--

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local player = Players.LocalPlayer

local flying = false
local speed = 50
local toggleKey = Enum.KeyCode.X
local bodyVelocity, bodyGyro
local currentAnimTrack
local idleAnimTrack

-- IDs de animaciones de vuelo en movimiento y posturas flotantes "pro" (Idle)
local anims = {
    {name = "Estilo Pro / Aura", moveId = "rbxassetid://507768375", idleId = "rbxassetid://507768434"},
    {name = "Postura Amenazante", moveId = "rbxassetid://507768388", idleId = "rbxassetid://507768375"}
}
local currentSelection = anims[1]

-- Interfaz Gráfica (GUI) del Panel
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "ProFlightGui"
screenGui.ResetOnSpawn = false
screenGui.Parent = player:WaitForChild("PlayerGui")

local mainFrame = Instance.new("Frame")
mainFrame.Size = UDim2.new(0, 260, 0, 310)
mainFrame.Position = UDim2.new(0.05, 0, 0.3, 0)
mainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
mainFrame.BorderSizePixel = 0
mainFrame.Active = true
mainFrame.Draggable = true
mainFrame.Parent = screenGui

local uiCorner = Instance.new("UICorner")
uiCorner.CornerRadius = UDim.new(0, 8)
uiCorner.Parent = mainFrame

local titleLabel = Instance.new("TextLabel")
titleLabel.Size = UDim2.new(1, 0, 0, 40)
titleLabel.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
titleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
titleLabel.TextSize = 15
titleLabel.Font = Enum.Font.SourceSansBold
titleLabel.Text = "Panel Pro Vuelo - JJS"
titleLabel.Parent = mainFrame

local titleCorner = Instance.new("UICorner")
titleCorner.CornerRadius = UDim.new(0, 8)
titleCorner.Parent = titleLabel

local function playAnimation(id)
    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        local animator = humanoid:FindFirstChildOfClass("Animator") or Instance.new("Animator", humanoid)
        if currentAnimTrack then currentAnimTrack:Stop() end
        local anim = Instance.new("Animation")
        anim.AnimationId = id
        currentAnimTrack = animator:LoadAnimation(anim)
        currentAnimTrack:Play()
    end
end

local function startFlight()
    local character = player.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    local rootPart = character.HumanoidRootPart
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    
    flying = true
    humanoid.PlatformStand = true

    playAnimation(currentSelection.moveId)

    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.Velocity = Vector3.new(0, 0, 0)
    bodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bodyVelocity.Parent = rootPart

    bodyGyro = Instance.new("BodyGyro")
    bodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bodyGyro.CFrame = rootPart.CFrame
    bodyGyro.Parent = rootPart

    task.spawn(function()
        local isMovingPrev = false
        while flying and character and character:Parent() do
            local camera = workspace.CurrentCamera
            local moveDir = Vector3.new(0, 0, 0)
            local moving = false
            
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + camera.CFrame.LookVector; moving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - camera.CFrame.LookVector; moving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - camera.CFrame.RightVector; moving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + camera.CFrame.RightVector; moving = true end

            bodyVelocity.Velocity = moveDir * speed
            bodyGyro.CFrame = camera.CFrame

            -- Cambia entre animación de movimiento o postura flotante "pro" si estás detenido
            if moving ~= isMovingPrev then
                isMovingPrev = moving
                if moving then
                    playAnimation(currentSelection.moveId)
                else
                    playAnimation(currentSelection.idleId) -- Posición estática pro flotando
                end
            end

            RunService.RenderStepped:Wait()
        end
    end)
end

local function stopFlight()
    flying = false
    local character = player.Character
    if character then
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then humanoid.PlatformStand = false end
    end
    if bodyVelocity then bodyVelocity:Destroy() end
    if bodyGyro then bodyGyro:Destroy() end
    if currentAnimTrack then currentAnimTrack:Stop() end
end

-- Botón principal en la GUI
local toggleBtn = Instance.new("TextButton")
toggleBtn.Size = UDim2.new(0.9, 0, 0, 35)
toggleBtn.Position = UDim2.new(0.05, 0, 0, 55)
toggleBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
toggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
toggleBtn.TextSize = 14
toggleBtn.Font = Enum.Font.SourceSansBold
toggleBtn.Text = "Activar Vuelo (Tecla X)"
toggleBtn.Parent = mainFrame

toggleBtn.MouseButton1Click:Connect(function()
    if flying then
        stopFlight()
        toggleBtn.Text = "Activar Vuelo"
    else
        startFlight()
        toggleBtn.Text = "Desactivar Vuelo"
    end
end)

-- Selector de Estilos Pro
local yOffset = 100
for _, data in ipairs(anims) do
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.9, 0, 0, 30)
    btn.Position = UDim2.new(0.05, 0, 0, yOffset)
    btn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextSize = 12
    btn.Font = Enum.Font.SourceSans
    btn.Text = data.name
    btn.Parent = mainFrame
    
    btn.MouseButton1Click:Connect(function()
        currentSelection = data
        if flying then
            stopFlight()
            startFlight()
        end
    end)
    yOffset = yOffset + 35
end

-- Controles de Velocidad en la GUI
local speedLabel = Instance.new("TextLabel")
speedLabel.Size = UDim2.new(0.9, 0, 0, 20)
speedLabel.Position = UDim2.new(0.05, 0, 0, 215)
speedLabel.BackgroundTransparency = 1
speedLabel.TextColor3 = Color3.fromRGB(200, 200, 200)
speedLabel.TextSize = 12
speedLabel.Font = Enum.Font.SourceSansBold
speedLabel.Text = "Velocidad: " .. speed
speedLabel.Parent = mainFrame

local minusBtn = Instance.new("TextButton")
minusBtn.Size = UDim2.new(0.42, 0, 0, 30)
minusBtn.Position = UDim2.new(0.05, 0, 0, 245)
minusBtn.BackgroundColor3 = Color3.fromRGB(150, 40, 40)
minusBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
minusBtn.Text = "Vel -"
minusBtn.Parent = mainFrame

minusBtn.MouseButton1Click:Connect(function()
    speed = math.max(10, speed - 10)
    speedLabel.Text = "Velocidad: " .. speed
end)

local plusBtn = Instance.new("TextButton")
plusBtn.Size = UDim2.new(0.42, 0, 0, 30)
plusBtn.Position = UDim2.new(0.53, 0, 0, 245)
plusBtn.BackgroundColor3 = Color3.fromRGB(40, 130, 40)
plusBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
plusBtn.Text = "Vel +"
plusBtn.Parent = mainFrame

plusBtn.MouseButton1Click:Connect(function()
    speed = speed + 10
    speedLabel.Text = "Velocidad: " .. speed
end)

-- Tecla rápida X
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == toggleKey then
        if flying then stopFlight() toggleBtn.Text = "Activar Vuelo" else startFlight() toggleBtn.Text = "Desactivar Vuelo" end
    end
end)

print("[+] Panel Pro con Loadstring listo.")
